#  Author :            Siddhant Vikas Balwadkar

############################################################################################
def Display(FileName):

    if(os.path.exists(FileName)):
        with open(FileName,"r") as file:
            for line in file:
                print(line, end="")
    else:
        print(f"File {FileName} does not exist.")
